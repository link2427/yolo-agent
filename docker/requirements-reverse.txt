# yolo-agent-reverse-engineering — extra packages layered onto the SAME
# /opt/pyenv environment defined by requirements-common.txt. There is exactly
# one Python 3.11 environment in the reverse image; this file only adds to it.
#
# Scope: .exe / .pyc extraction and Python bytecode decompilation, plus native
# binary inspection and emulation.
#
# Installed after the common set, because a couple of these must win a shared
# dependency. Read the notes before changing anything.
#
# THE ONE HARD CONSTRAINT: xdis is pinned to exactly 6.3.0, because
# pyinstxtractor-ng declares `xdis==6.3.0`. Extraction is the non-negotiable
# capability here — without it there is nothing to decompile — so every other
# pin has to live with that.
#
#   * decompyle3 is NOT installed. It declares `xdis<6.3`, which is genuinely
#     unsatisfiable alongside xdis==6.3.0. pip fails the whole install with
#     ResolutionImpossible rather than silently downgrading, which is how this
#     was caught. Do not add it back.
#   * uncompyle6 is present, transitively via pydumpck (pydumpck declares
#     `uncompyle6>=3.9.0`). Its own metadata also wants `xdis<6.2.0`, so pip
#     installs it without re-checking; the reverse smoke suite runs its CLI at
#     build time, which is what proves it actually works against xdis 6.3.0.
#     Because it is transitive, treat it as best-effort: if a future pydumpck
#     drops it, add an explicit pin here.
#   * pycdc + pycdas (built from source in the reverse image) are the
#     version-agnostic fallback and are always available.
#
# Every pin publishes a CPython 3.11 / linux-amd64 wheel; see PINS.md.

# --- .exe / PyInstaller extraction ------------------------------------------
# pyinstxtractor-ng unpacks PyInstaller executables and needs no execution of
# the target, which is what makes it safe on an air-gapped host.
pyinstxtractor-ng==2026.7.3
xdis==6.3.0

# --- Python bytecode decompilers --------------------------------------------
# pydumpck is the all-in-one orchestrator: it dispatches to whichever decompiler
# fits the bytecode version, and handles .exe/.pyz/.pyc/.elf inputs.
# It also pulls in PyInstaller and uncompyle6 as dependencies. Those are
# transitive, not declared: the image smoke test relies on PyInstaller to build a
# real archive, and runs uncompyle6 to verify it works. If pydumpck is ever
# dropped, add explicit pins for both here.
pydumpck==1.20.1

# --- native binary inspection / emulation -----------------------------------
capstone==5.0.9
unicorn==2.1.4
angr==9.2.213
lief==1.0.0
pyelftools==0.33
pefile==2024.8.26
