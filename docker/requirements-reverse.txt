# yolo-agent-reverse-engineering — extra packages layered onto the SAME
# /opt/pyenv environment defined by requirements-common.txt. There is exactly
# one Python 3.11 environment in the reverse image; this file only adds to it.
#
# Scope: .exe / .pyc extraction and Python bytecode decompilation, plus native
# binary inspection and emulation.
#
# Installed after the common set, because a couple of these must win a shared
# dependency. Read the notes before changing anything.
#
# THE ONE HARD CONSTRAINT: xdis is pinned to exactly 6.3.0, because
# pyinstxtractor-ng declares `xdis==6.3.0`. Extraction is the non-negotiable
# capability here — without it there is nothing to decompile — so every other
# pin has to live with that.
#
#   * decompyle3 is NOT installed. It declares `xdis<6.3`, which is genuinely
#     unsatisfiable alongside xdis==6.3.0. pip fails the whole install with
#     ResolutionImpossible rather than silently downgrading, which is how this
#     was caught. Do not add it back.
#   * uncompyle6 carries the SAME ceiling as decompyle3 (it declares
#     `xdis<6.3,>=6.1.0` in its wheel metadata), so it cannot be resolved
#     alongside xdis==6.3.0 either. pip is strict about transitive constraints,
#     so uncompyle6 and pydumpck are installed with --no-deps from
#     requirements-reverse-nodeps.txt. Their dependencies are pinned above, so
#     nothing is actually unpinned -- only the resolution edge is skipped, and
#     the reverse smoke suite runs uncompyle6's CLI at build time, which is the
#     real evidence that it works against xdis 6.3.0.
#   * pycdc + pycdas (built from source in the reverse image) are the
#     version-agnostic fallback and are always available.
#
# WHY angr IS ABSENT: its 22-package closure needs `mulpyplexer`, which
# publishes no wheel and no release compatible with Python 3.11 at all
# (pip reports "from versions: none"), and its `cle` dependency pins
# `arpy==1.1.1`, which is sdist-only. Rather than carry a source-built tail
# through an air-gapped image, angr was dropped. Every pin below now
# installs from a wheel, with no exceptions.
#
# Every pin publishes a CPython 3.11 / linux-amd64 wheel; see PINS.md.

# --- .exe / PyInstaller extraction ------------------------------------------
# pyinstxtractor-ng unpacks PyInstaller executables and needs no execution of
# the target, which is what makes it safe on an air-gapped host.
pyinstxtractor-ng==2026.7.3
xdis==6.3.0

# --- Python bytecode decompilers --------------------------------------------
# pydumpck is the all-in-one orchestrator (it dispatches to whichever decompiler
# fits the bytecode version and handles .exe/.pyz/.pyc/.elf inputs), but it is
# installed with --no-deps from requirements-reverse-nodeps.txt, because one of
# its dependencies is unresolvable here:
#
#   pydumpck -> uncompyle6 -> xdis<6.3      (all of 3.9.0 ... 3.9.3)
#   pyinstxtractor-ng      -> xdis==6.3.0   (exact)
#
# pip treats a transitive constraint as hard, so the two cannot be resolved
# together. pydumpck's other dependencies are therefore pinned here directly, so
# the environment is still complete and fully pinned -- only the resolution edge
# from pydumpck to them is skipped.
pyinstaller==6.22.2
sgtpyutils==1.20.15
tinyaes==1.1.2
# uncompyle6's remaining dependency (click already comes from the common set).
# 1.8.9 is the only release with a wheel and it satisfies uncompyle6's
# <1.9.2,>=1.8.9 range.
spark-parser==1.8.9

# --- native binary inspection / emulation -----------------------------------
# angr used to live here. It is deliberately absent -- see the note at the top
# and PINS.md. What remains is self-sufficient: none of these depend on angr,
# and the disassembler plus emulator pair carries scripted analysis.
capstone==5.0.9
unicorn==2.1.4
lief==1.0.0
pyelftools==0.33
pefile==2024.8.26
